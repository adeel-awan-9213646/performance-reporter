Dynamic Column Rendering & Filtering Logic function renderAgentTable(agentsData, selectedTeam) { // Filter agents by team tag let filteredAgents = agentsData.filter(agent => { if (selectedTeam === 'All Teams') return true; return agent.name.toLowerCase().includes(selectedTeam.toLowerCase()); }); // 2. Sort by Total Calls / Performance High to Low filteredAgents.sort((a, b) => b.totalCalls - a.totalCalls); // 3. Find active Dispositions (D1 to D9) where sale > 0 let activeDispositions = []; for (let i = 1; i <= 9; i++) { let dKey = 'D' + i; let hasSales = filteredAgents.some(agent => agent[dKey] && agent[dKey] > 0); if (hasSales) { activeDispositions.push(dKey); } } // 4. Determine Max/Min for Color Thresholds let maxCalls = Math.max(...filteredAgents.map(a => a.totalCalls)); // Generate Table Headers let headerHTML = ` Agent Name Agent ID Total Calls`; activeDispositions.forEach(d => { headerHTML += `${d}`; }); headerHTML += ``; // Generate Table Rows with Editable Fields & Color Coding let rowsHTML = filteredAgents.map(agent => { // Color Assignment Logic let rowClass = 'bg-yellow'; // Default Average if (agent.totalCalls >= maxCalls * 0.8) { rowClass = 'bg-green'; // Top Performer } else if (agent.totalCalls < maxCalls * 0.4) { rowClass = 'bg-red'; // Low Performer } let row = ` ${agent.name} ${agent.id} ${agent.totalCalls}`; activeDispositions.forEach(d => { row += `${agent[d] || 0}`; }); row += ``; return row; }).join(''); document.getElementById('tableHeader').innerHTML = headerHTML; document.getElementById('tableBody').innerHTML = rowsHTML; }
